hostname -f

rm /path/to/existing/keystore.p12
# or wherever WebLogic is pointing — check config.xml
grep -i "keystore" $DOMAIN_HOME/config/config.xml

keytool -genkeypair \
  -alias $(hostname -f) \
  -keyalg RSA \
  -keysize 2048 \
  -validity 365 \
  -storetype PKCS12 \
  -keystore $DOMAIN_HOME/security/identity.p12 \
  -storepass Welcome1 \
  -keypass Welcome1 \
  -dname "CN=$(hostname -f), OU=IT, O=MyOrg, L=City, ST=State, C=US" \
  -ext "SAN=dns:$(hostname -f),dns:$(hostname -s),ip:$(hostname -i)"
  
  
  # Export the cert from identity store
keytool -exportcert \
  -alias $(hostname -f) \
  -keystore $DOMAIN_HOME/security/identity.p12 \
  -storetype PKCS12 \
  -storepass Welcome1 \
  -file $DOMAIN_HOME/security/demo.crt

# Import into a trust keystore
keytool -importcert \
  -alias $(hostname -f) \
  -file $DOMAIN_HOME/security/demo.crt \
  -keystore $DOMAIN_HOME/security/trust.p12 \
  -storetype PKCS12 \
  -storepass Welcome1 \
  -noprompt
  
  In $DOMAIN_HOME/config/config.xml, find the <ssl> block and update:
  
  <keystore>Custom Identity and Custom Trust</keystore>
<custom-identity-key-store-file-name>/path/to/identity.p12</custom-identity-key-store-file-name>
<custom-identity-key-store-type>PKCS12</custom-identity-key-store-type>
<custom-identity-key-store-pass-phrase-encrypted>Welcome1</custom-identity-key-store-pass-phrase-encrypted>
<custom-trust-key-store-file-name>/path/to/trust.p12</custom-trust-key-store-file-name>
<custom-trust-key-store-type>PKCS12</custom-trust-key-store-type>
<custom-trust-key-store-pass-phrase-encrypted>Welcome1</custom-trust-key-store-pass-phrase-encrypted>

Sub CreateXMLFiles()
    Dim ws As Worksheet
    Dim lastRow As Long
    Dim i As Long
    Dim folderPath As String
    Dim fileName As String
    Dim fileNum As Integer
    Dim clobData As String
    
    Set ws = ThisWorkbook.Sheets(1)          ' Change if data is on another sheet
    
    ' === Change this path if you want ===
    folderPath = "C:\Temp\XML_Files\"        ' Folder where XML files will be saved
    
    ' Create folder if it doesn't exist
    If Dir(folderPath, vbDirectory) = "" Then
        MkDir folderPath
    End If
    
    lastRow = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row
    
    For i = 2 To lastRow                     ' Assuming row 1 is header
        fileName = folderPath & ws.Cells(i, 1).Value & ".xml"   ' Column A = ID
        clobData = ws.Cells(i, 2).Value                          ' Column B = CLOB
        
        ' Write the file
        fileNum = FreeFile
        Open fileName For Output As #fileNum
        Print #fileNum, clobData
        Close #fileNum
    Next i
    
    MsgBox "Done! XML files created in: " & folderPath
End Sub
